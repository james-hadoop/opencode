展示串行与并行编排的组合使用，模拟真实业务场景

flow.xml - LiteFlow 规则配置

// 串行编排：按顺序执行
THEN(
    validateOrder,      // 验证订单
    calculatePrice,     // 计算价格
    createOrder        // 创建订单
)

// 条件编排：根据条件选择分支
SWITCH(checkUserType)
    .TO(premiumFlow, "tag:premium")
    .TO(normalFlow, "tag:normal")

// 并行编排：同时执行多个任务
WHEN(
    sendEmail,         // 发送邮件
    sendSMS,           // 发送短信
    updateInventory   // 更新库存
)

// 循环编排：重复执行
LOOP(retryPayment).MAX(3)
